cmake_minimum_required(VERSION 3.20)
project(SkynetHostCore LANGUAGES CXX)

add_library(skynet_host_core STATIC src/workspace.cpp src/workspace_c.cpp)
target_compile_features(skynet_host_core PUBLIC cxx_std_17)
target_include_directories(skynet_host_core PUBLIC include PRIVATE private)
if(MSVC)
    target_compile_options(skynet_host_core PRIVATE /W4 /WX /EHsc)
else()
    target_compile_options(skynet_host_core PRIVATE -Wall -Wextra -Werror)
endif()

include(CTest)
if(BUILD_TESTING)
    add_executable(workspace_tests tests/workspace_test.cpp)
    target_include_directories(workspace_tests PRIVATE private)
    target_link_libraries(workspace_tests PRIVATE skynet_host_core)
    add_test(NAME workspace_transitions COMMAND workspace_tests)
endif()
